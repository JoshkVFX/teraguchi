    ```python
    """Shared structlog configuration (OBS-01).

    Imported by all three entry points (server, client, broker) and by the
    smoke harness. Idempotent — safe to call configure() multiple times.

    Schema per RESEARCH §"structlog schema" lines 856-895:
      Required fields (on every event):
        - ts     (ISO 8601 UTC)           via TimeStamper(fmt="iso", utc=True, key="ts")
        - level  (debug/info/warning/..)  via add_log_level
        - tier   (server/client/broker)   via contextvars bind at configure(tier=...)
        - event  (dotted short name)      via the first positional arg to log.info(...)

      Optional (bind per-context):
        session_id, client_id, username, stage, state, transport, latency_ms,
        frame_size, keyframe, err_class, err_code.

    Redaction (T-1-04): any event dict key matching REDACT_KEYS (case-insensitive
    substring match) has its VALUE replaced with "[REDACTED]" BEFORE JSONRenderer.
    """
    from __future__ import annotations

    import contextvars
    import logging
    import sys
    from typing import Any, Mapping

    import structlog
    from structlog.typing import EventDict, Processor, WrappedLogger

    # session_id contextvar (bound per-SessionRuntime in server; per-tab in client)
    session_id_var: contextvars.ContextVar[str] = contextvars.ContextVar(
        "session_id", default="")

    # Case-insensitive substring match — catches "password", "api_key", "TOKEN",
    # "user_credentials", "broker_secret", "private_key", "pin_code", etc.
    REDACT_KEYS: tuple[str, ...] = (
        "password", "credential", "token", "secret", "key", "pin",
    )


    def _redact_secrets(
        logger: WrappedLogger,
        method_name: str,
        event_dict: EventDict,
    ) -> EventDict:
        """Processor — strips any key containing a REDACT_KEYS substring.

        Runs BEFORE JSONRenderer per RESEARCH Pitfall 2. Replaces values with
        the literal string "[REDACTED]" (never drops the key — visibility that
        redaction happened is itself useful signal).
        """
        out: dict[str, Any] = {}
        for k, v in event_dict.items():
            lowered = str(k).lower()
            if any(needle in lowered for needle in REDACT_KEYS):
                out[k] = "[REDACTED]"
            else:
                out[k] = v
        return out


    _configured = False


    def configure(tier: str, *, verbose: bool = False) -> None:
        """Configure structlog once.

        Args:
            tier: 'server', 'client', or 'broker'. Bound into contextvars so
                every event carries it. Entry points call this in main().
            verbose: DEBUG level if True, else INFO.
        """
        global _configured
        if _configured:
            return
        _configured = True

        level = logging.DEBUG if verbose else logging.INFO
        processors: list[Processor] = [
            structlog.contextvars.merge_contextvars,        # MUST be before JSONRenderer (Pitfall 2)
            structlog.processors.add_log_level,
            structlog.processors.TimeStamper(fmt="iso", utc=True, key="ts"),
            structlog.processors.StackInfoRenderer(),
            structlog.processors.format_exc_info,
            structlog.processors.UnicodeDecoder(),
            _redact_secrets,                                # T-1-04 — before renderer
            structlog.processors.JSONRenderer(sort_keys=True),
        ]
        structlog.configure(
            processors=processors,
            wrapper_class=structlog.make_filtering_bound_logger(level),
            context_class=dict,
            logger_factory=structlog.PrintLoggerFactory(file=sys.stderr),
            cache_logger_on_first_use=True,
        )
        # Bridge stdlib logging.getLogger(...) → structlog so existing logger.info()
        # call sites (documented in CONVENTIONS.md §"Logging") flow through the same pipeline.
        logging.basicConfig(
            format="%(message)s",
            stream=sys.stderr,
            level=level,
        )
        # Tag every emit with the tier.
        structlog.contextvars.bind_contextvars(tier=tier)


    def get_logger(name: str = "") -> Any:
        """Return a structlog BoundLogger.

        Use `name="server.broadcaster"` so logs get `logger=teraguchi.server.broadcaster`.
        Call sites that still use stdlib logging get the same output via basicConfig bridge.
        """
        return structlog.get_logger(f"teraguchi.{name}" if name else "teraguchi")


    class StageTimer:
        """Context manager that binds `stage` + `latency_ms` for the block.

        Usage:
            with StageTimer("capture"):
                raw = capture.capture_raw_bgra()

        On exit, logs a stage.timing event with the elapsed ms.
        """
        def __init__(self, stage: str):
            self.stage = stage
            self._log = get_logger("stage")

        def __enter__(self):
            import time
            self._t0 = time.perf_counter()
            structlog.contextvars.bind_contextvars(stage=self.stage)
            return self

        def __exit__(self, exc_type, exc, tb):
            import time
            elapsed_ms = (time.perf_counter() - self._t0) * 1000.0
            self._log.info("stage.timing", stage=self.stage, latency_ms=round(elapsed_ms, 3))
            structlog.contextvars.unbind_contextvars("stage")
            return False   # don't suppress
    ```

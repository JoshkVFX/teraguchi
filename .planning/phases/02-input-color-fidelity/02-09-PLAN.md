Step D -- periodic safety net. Add to `server/client_session.py` (or `SessionRuntime`):

```
async def _modifier_periodic_safety_loop(self):
    import time
    while not self._stopped:
        await asyncio.sleep(2)
        now = time.monotonic()
        last = self._last_key_event_at  # set in the KEY_EVENT dispatch
        last_had_modifiers = self._last_key_event_had_modifiers
        if last is None:
            continue
        if last_had_modifiers:
            continue  # don't break a held chord
        if (now - last) >= 10.0:
            self._input_injector.reset_modifiers()
            log.info("input.periodic_reset_modifiers", quiet_seconds=(now - last))
            self._last_key_event_at = now  # reset the countdown
```

    ```python
    """STAB-05 characterization — server bootstraps + handshake reaches streaming.

    Lands BEFORE Plans 10-11's module extraction. Plans 10-11 do NOT modify
    this file. If their code-move breaks this test, the extraction is wrong.
    """
    import asyncio
    import json
    import ssl
    from unittest import mock

    import pytest
    import websockets

    from common.messages import (AuthRequest, AuthResponse, AuthResult,
                                   ClientHelloMsg, MsgType, parse_message)


    @pytest.fixture
    def mock_encoder_subprocess(monkeypatch):
        """D-02 — mock ffmpeg subprocess boundary. VideoEncoder.start() Popen calls
        return a fake process that never writes real encoded frames."""
        mock_proc = mock.MagicMock()
        mock_proc.stdin = mock.MagicMock()
        mock_proc.stdout = mock.MagicMock()
        mock_proc.stdout.read.return_value = b""
        mock_proc.poll.return_value = None
        mock_proc.returncode = None
        monkeypatch.setattr("subprocess.Popen", lambda *a, **kw: mock_proc)
        return mock_proc


    @pytest.mark.asyncio
    async def test_server_bootstraps_and_accepts_client(
        tls_ca_and_cert, free_port, mock_encoder_subprocess, monkeypatch
    ):
        """End-to-end: server accepts, client authenticates (mode=none), ServerHello arrives."""
        # Import late so the subprocess monkeypatch binds before SessionRuntime imports
        from server.main import SessionRuntime, handle_client, Authenticator
        import server.main as srv_main

        # Set up global auth in "none" mode (skips challenge/response)
        monkeypatch.setattr(srv_main, "auth", Authenticator(mode="none"))
        monkeypatch.setattr(srv_main, "runtimes", {})
        monkeypatch.setattr(srv_main, "default_runtime", None)

        # Start TLS WSS server using the test CA
        srv_ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
        srv_ctx.load_cert_chain(str(tls_ca_and_cert["server_cert"]),
                                 str(tls_ca_and_cert["server_key"]))
        srv_ctx.minimum_version = ssl.TLSVersion.TLSv1_2

        async def run_handler(ws):
            # Let handle_client run until the client closes
            try:
                await handle_client(ws)
            except Exception:
                pass

        async with websockets.serve(run_handler, "127.0.0.1", free_port, ssl=srv_ctx):
            cli_ctx = ssl.create_default_context(
                cafile=str(tls_ca_and_cert["ca_cert"]))
            async with websockets.connect(f"wss://127.0.0.1:{free_port}",
                                           ssl=cli_ctx) as ws:
                # Read first message — either AuthRequest (if auth enabled) or
                # ServerHello (if auth=none). mode='none' → ServerHello directly.
                first = await asyncio.wait_for(ws.recv(), timeout=5.0)
                msg = parse_message(first)
                assert msg.get("type") in (MsgType.AUTH_REQUEST, MsgType.SERVER_HELLO)

                # If auth requested, send a no-op response
                if msg.get("type") == MsgType.AUTH_REQUEST:
                    await ws.send(AuthResponse(username="", hash="",
                                                 challenge_response="").to_json())
                    # Read next — either AuthResult or ServerHello
                    next_msg = await asyncio.wait_for(ws.recv(), timeout=5.0)
                    parsed = parse_message(next_msg)
                    assert parsed.get("type") in (MsgType.AUTH_RESULT,
                                                    MsgType.SERVER_HELLO)

                # Send ClientHello (complete handshake)
                await ws.send(ClientHelloMsg(screen_width=1920,
                                               screen_height=1080).to_json())
                # Give the server a moment to transition FSM
                await asyncio.sleep(0.2)


    @pytest.mark.asyncio
    async def test_server_fsm_reaches_streaming_on_hello(
        tls_ca_and_cert, free_port, mock_encoder_subprocess, monkeypatch
    ):
        """STAB-06 regression: after ClientHello, the server-side per-client FSM

    ```
